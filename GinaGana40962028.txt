use movieco;

CREATE TABLE MEMBERSHIP (
    MEM_NUM INT PRIMARY KEY NOT NULL,
    MEM_FNAME VARCHAR(20),
    MEM_LNAME VARCHAR(30),
    MEM_STREET VARCHAR(50),
    MEM_CITY VARCHAR(30),
    MEM_STATE CHAR(2),   
    MEM_ZIP CHAR(5),
    MEM_BALANCE INT
);

CREATE TABLE PRICE (
    PRICE_CODE INT PRIMARY KEY NOT NULL,
    PRICE_DESCRIPTION VARCHAR(20),
    PRICE_RENTFEE DECIMAL(4,1),
    PRICE_DAILYLATEFEE DECIMAL(4,1)
);

CREATE TABLE RENTAL (
    RENT_NUM INT PRIMARY KEY NOT NULL,
    RENT_DATE VARCHAR(15),
    MEM_NUM INT NOT NULL,
    FOREIGN KEY(MEM_NUM) REFERENCES MEMBERSHIP(MEM_NUM)
);

CREATE TABLE MOVIE (
    MOVIE_NUM INT PRIMARY KEY NOT NULL,
    MOVIE_TITLE VARCHAR(100),
    MOVIE_YEAR  SMALLINT,      -- 2bytes
    MOVIE_COST  DECIMAL(5,2),
    MOVIE_GENRE VARCHAR(20),
    PRICE_CODE  INT,
  FOREIGN KEY(PRICE_CODE) REFERENCES PRICE(PRICE_CODE)
);

CREATE TABLE VIDEO (
    VID_NUM INT PRIMARY KEY NOT NULL,
    VID_INDATE VARCHAR(15),
    MOVIE_NUM INT NOT NULL,
	FOREIGN KEY(MOVIE_NUM) REFERENCES MOVIE(MOVIE_NUM)
);

CREATE TABLE DETAILRENTAL (
    RENT_NUM INT NOT NULL,
    VID_NUM INT NOT NULL,
    DETAIL_FEE DECIMAL(4,1),
    DETAIL_DUEDATE VARCHAR(15),
    DETAIL_RETURNDATE VARCHAR(15),
    DETAIL_DAILYLATEFEE INT,
    PRIMARY KEY (RENT_NUM, VID_NUM),
	FOREIGN KEY (RENT_NUM) REFERENCES RENTAL (RENT_NUM),
	FOREIGN KEY (VID_NUM) REFERENCES VIDEO (VID_NUM)
);
